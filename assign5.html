import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

# 1. Create the dataset
data = pd.DataFrame({
    'SquareFootage': [1000, 1500, 2000, 2500, 3000, 3500],
    'HousePrice': [150000, 200000, 250000, 280000, 350000, 400000]
})

print("Dataset preview:")
print(data.head())

# 2. Independent (X) and dependent (y) variables
X = data[['SquareFootage']]
y = data['HousePrice']

# 3. Train the Linear Regression model
model = LinearRegression()
model.fit(X, y)

# 4. Get regression coefficients
intercept = float(model.intercept_)
slope = float(model.coef_[0])

print(f"\nRegression Equation:")
print(f"Price = {intercept:.2f} + {slope:.2f} × SquareFootage")

# 5. Make predictions
y_pred = model.predict(X)

print("\nPredicted House Prices:")
print(y_pred)

# 6. Calculate MSE and R²
mse = mean_squared_error(y, y_pred)
r2 = r2_score(y, y_pred)

print(f"\nMean Squared Error (MSE): {mse:.2f}")
print(f"R² Score: {r2:.4f}")

# 7. Plot the data and regression line
plt.scatter(X, y, label='Actual House Prices')
plt.plot(X, y_pred, label='Regression Line')

plt.xlabel('Square Footage')
plt.ylabel('House Price')
plt.title('Linear Regression: House Price vs Square Footage')
plt.legend()
plt.show()
